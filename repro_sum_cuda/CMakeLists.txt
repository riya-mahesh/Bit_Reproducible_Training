cmake_minimum_required(VERSION 3.24)
project(repro_sum_cuda LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES native)
add_library(repro_sum repro_sum.cu)
target_compile_options(repro_sum PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
target_include_directories(repro_sum PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(test_repro_sum test_repro_sum.cu)
target_link_libraries(test_repro_sum PRIVATE repro_sum)
add_executable(bench_repro_sum bench_repro_sum.cu)
target_link_libraries(bench_repro_sum PRIVATE repro_sum)
